# anydeck backend dependencies.
#
# Setup:
#   python3 -m venv .venv
#   .venv/bin/python -m pip install -r requirements.txt

# HTTP API. Bound to 127.0.0.1 only, never 0.0.0.0 (see requirements.md).
fastapi==0.141.1
uvicorn[standard]==0.52.3

# Communication with the input devices over vendor-defined HID.
# This is cython-hidapi, the Python binding of the hidapi C library; it is
# imported as `import hid` and ships the native library in its wheels.
# (Not to be confused with the PyPI package literally named "hid", a ctypes
# wrapper that requires a system-wide libhidapi.)
hidapi==0.15.0

# Standalone application window instead of a browser tab.
#
# pywebview is only a thin Python wrapper - the actual rendering is done by a
# system GUI toolkit. Windows (WebView2) and macOS (WKWebView) ship one, so pip
# alone is enough there. On Linux it has to be installed as a SYSTEM package,
# pip cannot provide it:
#   Arch:   sudo pacman -S --needed webkit2gtk-4.1 python-gobject python-cairo
#   Ubuntu: sudo apt install gir1.2-webkit2-4.1 python3-gi python3-gi-cairo
# scripts/dev.py checks for this and prints the command if it is missing.
pywebview==6.2.1

# Background tray icon (pillow renders the icon image).
pystray==0.19.5
pillow==12.3.0

# The action mapping is stored in SQLite through the sqlite3 module of the
# Python standard library, so it needs no separate dependency.

# Stored credentials: the key lives in the operating system's own credential
# store (macOS Keychain, Windows Credential Manager, Linux SecretService), never
# in the database next to the data it protects.
#
# On Linux this needs a running secret service - gnome-keyring or KWallet over
# D-Bus. There is none on a bare window manager or over SSH, and utils/crypto.py
# says so rather than failing at the first decrypt.
#
# PACKAGING: keyring finds its backends through entry points, which PyInstaller
# does not follow. packaging/anydeck.spec must collect them explicitly:
#   from PyInstaller.utils.hooks import collect_all
#   datas, binaries, hiddenimports = collect_all("keyring")
# Without it the bundled app starts fine and fails the moment a secret is saved.
keyring==25.7.0

# AES-256-GCM for the encrypted half of a secrets row. Authenticated, so a
# tampered or truncated value fails loudly instead of decrypting to garbage.
# The standard library has no AES at all, hence a dependency rather than hashlib.
cryptography==50.0.0
