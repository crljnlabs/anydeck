# Communication with the input devices over vendor-defined HID.
# This is cython-hidapi, the Python binding of the hidapi C library; it is
# imported as `import hid` and ships the native library in its wheels.
# (Not to be confused with the PyPI package literally named "hid", a ctypes
# wrapper that requires a system-wide libhidapi.)
hidapi==0.15.0

# Standalone application window instead of a browser tab.
#
# pywebview is only a thin Python wrapper - the actual rendering is done by a
# system GUI toolkit. Windows (WebView2) and macOS (WKWebView) ship one, so pip
# alone is enough there. On Linux it has to be installed as a SYSTEM package,
# pip cannot provide it:
#   Arch:   sudo pacman -S --needed webkit2gtk-4.1 python-gobject python-cairo
#   Ubuntu: sudo apt install gir1.2-webkit2-4.1 python3-gi python3-gi-cairo
# scripts/dev.py checks for this and prints the command if it is missing.
pywebview==6.2.1

# Background tray icon (pillow renders the icon image).
pystray==0.19.5
pillow==12.3.0
